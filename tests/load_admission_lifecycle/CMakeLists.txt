cmake_minimum_required(VERSION 3.21)
project(LoadAdmissionLifetime LANGUAGES CXX)
enable_testing()
set(source_file "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/LoadAdmissionRuntime.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_file}")
file(READ "${source_file}" source)
string(FIND "${source}" "std::uint64_t NextEvent() noexcept" begin)
if(begin LESS 0)
  message(FATAL_ERROR "Runtime lifecycle boundary missing")
endif()
string(SUBSTRING "${source}" ${begin} -1 body)
string(REGEX REPLACE "\n}[ \t\r\n]*$" "\n" body "${body}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lifecycle.inc" "${body}")
add_executable(load_admission_lifecycle test.cpp)
find_package(Threads REQUIRED)
target_link_libraries(load_admission_lifecycle PRIVATE Threads::Threads)
set_target_properties(load_admission_lifecycle PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(load_admission_lifecycle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME actual_runtime_lifecycle COMMAND load_admission_lifecycle)

# Compile actual inner-hook and reader preparation control flow with synthetic
# native callbacks, including same-thread reentry through the recursive lock.
set(hook_file "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/Hooks_SaveLoad.cpp")
set(serial_file "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/Serialization.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${hook_file}" "${serial_file}")
file(READ "${hook_file}" hook_source)
string(FIND "${hook_source}" "bool BGSSaveLoadManager::LoadGame_Hook(" hook_begin)
string(FIND "${hook_source}" "\nbool\t\ts_requestedSave" hook_end)
if(hook_begin LESS 0 OR hook_end LESS hook_begin)
  message(FATAL_ERROR "Inner hook extraction boundary missing")
endif()
math(EXPR hook_length "${hook_end}-${hook_begin}")
string(SUBSTRING "${hook_source}" ${hook_begin} ${hook_length} hook_body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/inner-hook.inc" "${hook_body}")
file(READ "${serial_file}" serial_source)
string(FIND "${serial_source}" "bool s_admittedFilePrepared = false;" serial_begin)
string(FIND "${serial_source}" "void SetSaveName(" serial_end)
if(serial_begin LESS 0 OR serial_end LESS serial_begin)
  message(FATAL_ERROR "Prepared reader extraction boundary missing")
endif()
math(EXPR serial_length "${serial_end}-${serial_begin}")
string(SUBSTRING "${serial_source}" ${serial_begin} ${serial_length} serial_body)
string(REPLACE "#endif" "" serial_body "${serial_body}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/prepare.inc" "${serial_body}")
add_executable(load_inner_lifecycle inner_test.cpp)
target_compile_definitions(load_inner_lifecycle PRIVATE ENSRICK_EXPERIMENTAL_SAVE_ADMISSION=1)
target_compile_features(load_inner_lifecycle PRIVATE cxx_std_11)
target_include_directories(load_inner_lifecycle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME actual_inner_reentry COMMAND load_inner_lifecycle)
