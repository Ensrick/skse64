cmake_minimum_required(VERSION 3.21)
project(SKSELoadArgumentRegression LANGUAGES CXX)
enable_testing()
set(source_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_dir}/Hooks_SaveLoad.cpp" "${source_dir}/GameData.h")
file(READ "${source_dir}/Hooks_SaveLoad.cpp" source)
string(FIND "${source}" "bool BGSSaveLoadManager::LoadGame_Hook(" begin)
string(FIND "${source}" "bool\t\ts_requestedSave" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Production function boundaries changed")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_load.inc" "${body}")
file(READ "${source_dir}/GameData.h" declarations)
string(REGEX MATCH "bool[ \t]+LoadGame_Hook\\([^;]+;" hook "${declarations}")
string(REGEX MATCH "DEFINE_MEMBER_FN\\(LoadGame_HookTarget,[^;]+;" target "${declarations}")
if(NOT hook OR NOT target)
  message(FATAL_ERROR "Production load declarations not found")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_declarations.inc" "${hook}\n${target}\n")
add_executable(load_arguments load_arguments.cpp)
target_compile_features(load_arguments PRIVATE cxx_std_20)
target_include_directories(load_arguments PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME production_load_argument_forwarding COMMAND load_arguments)
