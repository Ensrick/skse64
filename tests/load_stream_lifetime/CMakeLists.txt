project(LoadStreamLifetimeTests LANGUAGES CXX)
enable_testing()

set_target_properties(load_stream_lifetime PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
add_test(NAME bounded_diagnostic_forwarding COMMAND load_stream_lifetime)
set(source_file "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/LoadStreamLifetime.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_file}")
file(READ "${source_file}" source)
string(FIND "${source}" "template<std::size_t N>" begin)
string(FIND "${source}" "\n}\n\nbool Ready()" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Live detour verifier boundaries missing")
endif()
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${source}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/verify.inc" "${body}")
add_executable(lifetime_hook_verifier verifier_test.cpp)
target_compile_features(lifetime_hook_verifier PRIVATE cxx_std_11)
target_include_directories(lifetime_hook_verifier PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME live_hook_readback_contract COMMAND lifetime_hook_verifier)
