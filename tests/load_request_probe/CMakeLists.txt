cmake_minimum_required(VERSION 3.21)
project(LoadRequestProbeRegression LANGUAGES CXX)
enable_testing()
set(source_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_dir}/Hooks_SaveLoad.cpp" "${source_dir}/GameData.h")
file(READ "${source_dir}/Hooks_SaveLoad.cpp" source)
string(FIND "${source}" "bool BGSSaveLoadManager::LoadRequestProbe_Hook(" begin)
string(FIND "${source}" "void BGSSaveLoadManager::SaveGame_Hook(" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Production probe boundaries changed")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production.inc" "${body}")
file(READ "${source_dir}/GameData.h" declarations)
string(REGEX MATCH "bool[ \t]+LoadRequestProbe_Hook\\([^;]+;" hook "${declarations}")
string(REGEX MATCH "DEFINE_MEMBER_FN\\(LoadRequestProbe_Target,[^;]+;" target "${declarations}")
if(NOT hook OR NOT target)
  message(FATAL_ERROR "Production probe declarations absent")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/declarations.inc" "${hook}\n${target}\n")
add_executable(load_request_probe test.cpp)
target_compile_features(load_request_probe PRIVATE cxx_std_20)
target_include_directories(load_request_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME production_request_probe COMMAND load_request_probe)
