cmake_minimum_required(VERSION 3.16)
project(LoadPluginSnapshotRegression LANGUAGES CXX)
enable_testing()
set(header_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header_dir}/LoadPluginSnapshot.h")
if(NOT EXISTS "${header_dir}/LoadPluginSnapshot.h")
  message(FATAL_ERROR "skse64/LoadPluginSnapshot.h is missing")
endif()
add_executable(plugin_snapshot test.cpp)
target_compile_features(plugin_snapshot PRIVATE cxx_std_11)
target_include_directories(plugin_snapshot PRIVATE "${header_dir}")
if(MSVC)
  target_compile_options(plugin_snapshot PRIVATE /W4)
else()
  target_compile_options(plugin_snapshot PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME plugin_snapshot_synthetic COMMAND plugin_snapshot)
