cmake_minimum_required(VERSION 3.21)
project(AdmittedSnapshot LANGUAGES CXX)
enable_testing()
add_executable(admitted_snapshot test.cpp)
target_compile_features(admitted_snapshot PRIVATE cxx_std_14)
add_test(NAME production_snapshot_adapter COMMAND admitted_snapshot)
set(serialization "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/Serialization.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${serialization}" "${CMAKE_CURRENT_SOURCE_DIR}/test.cpp")
file(READ "${serialization}" source)
string(FIND "${source}" "bool WriteRecord(" begin)
string(FIND "${source}" "bool ResolveFormId(" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Serialization API boundaries missing")
endif()
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${source}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/api.inc" "${body}")
string(FIND "${source}" "void HandleSaveGlobalData(void)" begin)
string(SUBSTRING "${source}" ${begin} -1 save_body)
string(FIND "${save_body}" "\n\t\ttry" end)
if(begin LESS 0 OR end LESS 0)
  message(FATAL_ERROR "Save pre-deletion guard boundary missing")
endif()
string(SUBSTRING "${save_body}" 0 ${end} save_prefix)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/save-prefix.inc" "${save_prefix}\n}\n")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/test.cpp" fake)
string(FIND "${fake}" "struct File {" begin)
string(FIND "${fake}" "int main()" end)
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${fake}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fake-file.inc" "${body}")
add_executable(admitted_snapshot_api api_test.cpp)
target_compile_features(admitted_snapshot_api PRIVATE cxx_std_14)
target_compile_definitions(admitted_snapshot_api PRIVATE ENSRICK_EXPERIMENTAL_SAVE_ADMISSION=1)
target_include_directories(admitted_snapshot_api PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME production_serialization_api COMMAND admitted_snapshot_api)
