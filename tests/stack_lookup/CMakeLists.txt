cmake_minimum_required(VERSION 3.21)
project(SKSEStackLookupRegression LANGUAGES CXX)
enable_testing()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/PapyrusVM.cpp")
# Compile the actual production function, not a copied reimplementation. Mock
# only its map/lock boundary; the live-game probe verifies the real VM layout.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../skse64/PapyrusVM.cpp" source)
string(FIND "${source}" "VMStackInfo* VMClassRegistry::GetStackInfo(UInt32 stackId)" begin)
string(FIND "${source}" "UInt32 SkyrimVM::ClearInvalidRegistrations(void)" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Production function boundaries changed; review the fixture")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_lookup.inc" "${body}")
add_executable(stack_lookup stack_lookup.cpp)
target_include_directories(stack_lookup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(stack_lookup PRIVATE cxx_std_20)
add_test(NAME production_stack_lookup COMMAND stack_lookup)
